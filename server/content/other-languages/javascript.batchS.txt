//### daily-stock-span
function stockSpans(prices) {
  const stack = [], out = [];
  for (const p of prices) {
    let span = 1;
    while (stack.length > 0 && stack[stack.length - 1][0] <= p) span += stack.pop()[1];
    stack.push([p, span]);
    out.push(span);
  }
  return out;
}
//### next-greater-circular
function nextGreaterCircular(numbers) {
  const n = numbers.length, out = new Array(n).fill(-1), stack = [];
  for (let i = 0; i < 2 * n; i++) {
    const x = numbers[i % n];
    while (stack.length > 0 && numbers[stack[stack.length - 1]] < x) out[stack.pop()] = x;
    if (i < n) stack.push(i);
  }
  return out;
}
//### simplify-path
function simplifyPath(path) {
  const stack = [];
  for (const part of path.split("/")) {
    if (part === "" || part === ".") continue;
    if (part === "..") { if (stack.length > 0) stack.pop(); }
    else stack.push(part);
  }
  return "/" + stack.join("/");
}
//### min-stack-queries
function minStackAnswers(ops) {
  const values = [], mins = [], out = [];
  for (const op of ops) {
    if (op > 0) {
      values.push(op);
      mins.push(mins.length > 0 ? Math.min(op, mins[mins.length - 1]) : op);
    } else if (op === 0) {
      if (values.length > 0) { values.pop(); mins.pop(); }
    } else {
      out.push(mins.length > 0 ? mins[mins.length - 1] : 0);
    }
  }
  return out;
}
//### longest-valid-brackets
function longestValidBrackets(text) {
  const stack = [-1];
  let best = 0;
  for (let i = 0; i < text.length; i++) {
    if (text[i] === "(") stack.push(i);
    else {
      stack.pop();
      if (stack.length === 0) stack.push(i);
      else best = Math.max(best, i - stack[stack.length - 1]);
    }
  }
  return best;
}
//### calculator-with-brackets
function calculateExpression(expression) {
  let result = 0, number = 0, sign = 1;
  const stack = [];
  for (const ch of expression) {
    if (ch >= "0" && ch <= "9") number = number * 10 + (ch.charCodeAt(0) - 48);
    else if (ch === "+" || ch === "-") { result += sign * number; number = 0; sign = ch === "+" ? 1 : -1; }
    else if (ch === "(") { stack.push(result); stack.push(sign); result = 0; sign = 1; }
    else if (ch === ")") {
      result += sign * number;
      number = 0;
      result *= stack.pop();
      result += stack.pop();
    }
  }
  return result + sign * number;
}
//### dedupe-sorted-list
function removeDuplicatesSorted(head) {
  let cur = head;
  while (cur !== null && cur.next !== null) {
    if (cur.val === cur.next.val) cur.next = cur.next.next;
    else cur = cur.next;
  }
  return head;
}
//### partition-list-around
function partitionList(head, pivot) {
  const lessHead = new ListNode(0, null), moreHead = new ListNode(0, null);
  let less = lessHead, more = moreHead;
  while (head !== null) {
    if (head.val < pivot) { less.next = head; less = head; }
    else { more.next = head; more = head; }
    head = head.next;
  }
  more.next = null;
  less.next = moreHead.next;
  return lessHead.next;
}
//### rotate-list-right
function rotateList(head, k) {
  if (head === null || head.next === null) return head;
  let length = 1, tail = head;
  while (tail.next !== null) { tail = tail.next; length++; }
  k %= length;
  if (k === 0) return head;
  tail.next = head;
  let newTail = head;
  for (let i = 0; i < length - k - 1; i++) newTail = newTail.next;
  const newHead = newTail.next;
  newTail.next = null;
  return newHead;
}
//### swap-adjacent-nodes
function swapPairs(head) {
  const dummy = new ListNode(0, head);
  let prev = dummy;
  while (prev.next !== null && prev.next.next !== null) {
    const a = prev.next, b = a.next;
    a.next = b.next;
    b.next = a;
    prev.next = b;
    prev = a;
  }
  return dummy.next;
}
