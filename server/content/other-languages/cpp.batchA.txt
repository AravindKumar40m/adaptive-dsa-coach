//### missing-number
int find_missing(vector<int>& numbers) {
    long long sum = (long long) numbers.size() * (numbers.size() + 1) / 2;
    for (int x : numbers) sum -= x;
    return (int) sum;
}
//### product-except-self
vector<int> product_except_self(vector<int>& nums) {
    int n = (int) nums.size();
    vector<int> result(n, 1);
    int left = 1;
    for (int i = 0; i < n; i++) { result[i] = left; left *= nums[i]; }
    int right = 1;
    for (int i = n - 1; i >= 0; i--) { result[i] *= right; right *= nums[i]; }
    return result;
}
//### rotate-right
vector<int> rotate_right(vector<int>& nums, int k) {
    int n = (int) nums.size();
    k %= n;
    vector<int> out(n);
    for (int i = 0; i < n; i++) out[(i + k) % n] = nums[i];
    return out;
}
//### first-missing-positive
int first_missing_positive(vector<int>& numbers) {
    vector<int> a = numbers;
    int n = (int) a.size();
    for (int i = 0; i < n; i++) {
        while (a[i] >= 1 && a[i] <= n && a[a[i] - 1] != a[i]) swap(a[i], a[a[i] - 1]);
    }
    for (int i = 0; i < n; i++) if (a[i] != i + 1) return i + 1;
    return n + 1;
}
//### rain-catch
int trapped_water(vector<int>& heights) {
    int left = 0, right = (int) heights.size() - 1, left_max = 0, right_max = 0, total = 0;
    while (left < right) {
        if (heights[left] < heights[right]) {
            left_max = max(left_max, heights[left]);
            total += left_max - heights[left];
            left++;
        } else {
            right_max = max(right_max, heights[right]);
            total += right_max - heights[right];
            right--;
        }
    }
    return total;
}
//### is-anagram
bool is_anagram(string first, string second) {
    if (first.size() != second.size()) return false;
    int count[26] = {0};
    for (char ch : first) count[ch - 'a']++;
    for (char ch : second) if (--count[ch - 'a'] < 0) return false;
    return true;
}
//### compress-string
string compress_text(string text) {
    string out;
    size_t i = 0;
    while (i < text.size()) {
        size_t j = i;
        while (j < text.size() && text[j] == text[i]) j++;
        out += text[i];
        out += to_string(j - i);
        i = j;
    }
    return out;
}
//### word-pattern-match
bool follows_pattern(string pattern, string sentence) {
    vector<string> words;
    istringstream in(sentence);
    string w;
    while (in >> w) words.push_back(w);
    if (words.size() != pattern.size()) return false;
    map<char, string> letter_to_word;
    map<string, char> word_to_letter;
    for (size_t i = 0; i < pattern.size(); i++) {
        char letter = pattern[i];
        if (letter_to_word.count(letter) && letter_to_word[letter] != words[i]) return false;
        if (word_to_letter.count(words[i]) && word_to_letter[words[i]] != letter) return false;
        letter_to_word[letter] = words[i];
        word_to_letter[words[i]] = letter;
    }
    return true;
}
//### longest-palindrome-part
int longest_palindrome_length(string text) {
    int n = (int) text.size(), best = 1;
    for (int centre = 0; centre < 2 * n - 1; centre++) {
        int lo = centre / 2, hi = lo + centre % 2;
        while (lo >= 0 && hi < n && text[lo] == text[hi]) {
            best = max(best, hi - lo + 1);
            lo--; hi++;
        }
    }
    return best;
}
//### min-window-cover
int min_cover_length(string text, string need) {
    int count[26] = {0};
    for (char ch : need) count[ch - 'a']++;
    int missing = (int) need.size(), left = 0, best = 0;
    for (int right = 0; right < (int) text.size(); right++) {
        int idx = text[right] - 'a';
        if (count[idx] > 0) missing--;
        count[idx]--;
        while (missing == 0) {
            if (best == 0 || right - left + 1 < best) best = right - left + 1;
            int l = text[left] - 'a';
            count[l]++;
            if (count[l] > 0) missing++;
            left++;
        }
    }
    return best;
}
//### shortest-palindrome-extension
int chars_to_prepend(string text) {
    string rev(text.rbegin(), text.rend());
    string s = text + "#" + rev;
    vector<int> fail(s.size(), 0);
    for (size_t i = 1; i < s.size(); i++) {
        int k = fail[i - 1];
        while (k > 0 && s[i] != s[k]) k = fail[k - 1];
        if (s[i] == s[k]) k++;
        fail[i] = k;
    }
    return (int) text.size() - fail[s.size() - 1];
}
