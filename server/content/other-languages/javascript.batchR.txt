//### longest-harmonious-length
function longestHarmonious(numbers) {
  const counts = new Map();
  for (const x of numbers) counts.set(x, (counts.get(x) || 0) + 1);
  let best = 0;
  for (const [v, c] of counts) {
    if (counts.has(v + 1)) best = Math.max(best, c + counts.get(v + 1));
  }
  return best;
}
//### anagram-window-count
function countAnagramWindows(text, pattern) {
  const m = pattern.length;
  if (text.length < m) return 0;
  const need = new Array(26).fill(0), have = new Array(26).fill(0);
  for (const ch of pattern) need[ch.charCodeAt(0) - 97]++;
  for (let i = 0; i < m; i++) have[text.charCodeAt(i) - 97]++;
  const same = () => { for (let c = 0; c < 26; c++) if (need[c] !== have[c]) return false; return true; };
  let result = same() ? 1 : 0;
  for (let i = m; i < text.length; i++) {
    have[text.charCodeAt(i) - 97]++;
    have[text.charCodeAt(i - m) - 97]--;
    if (same()) result++;
  }
  return result;
}
//### unique-sum
function sumOfUnique(numbers) {
  const counts = new Map();
  for (const x of numbers) counts.set(x, (counts.get(x) || 0) + 1);
  let total = 0;
  for (const [v, c] of counts) if (c === 1) total += v;
  return total;
}
//### max-points-on-line
function maxPointsOnLine(points) {
  const gcd = (a, b) => (b === 0 ? a : gcd(b, a % b));
  let best = 1;
  for (let i = 0; i < points.length; i++) {
    const slopes = new Map();
    for (let j = i + 1; j < points.length; j++) {
      let dx = points[j][0] - points[i][0], dy = points[j][1] - points[i][1];
      const g = gcd(Math.abs(dx), Math.abs(dy));
      dx /= g;
      dy /= g;
      if (dx < 0 || (dx === 0 && dy < 0)) { dx = -dx; dy = -dy; }
      const key = dx + "," + dy;
      const c = (slopes.get(key) || 0) + 1;
      slopes.set(key, c);
      best = Math.max(best, c + 1);
    }
  }
  return best;
}
//### common-prefix-length
function commonPrefixLength(first, second, third) {
  let i = 0;
  while (i < first.length && i < second.length && i < third.length && first[i] === second[i] && second[i] === third[i]) i++;
  return i;
}
//### parse-integer-text
function parseInteger(text) {
  let i = 0;
  const n = text.length;
  while (i < n && text[i] === " ") i++;
  let sign = 1;
  if (i < n && (text[i] === "+" || text[i] === "-")) { sign = text[i] === "-" ? -1 : 1; i++; }
  let value = 0;
  while (i < n && text[i] >= "0" && text[i] <= "9") {
    value = value * 10 + (text.charCodeAt(i) - 48);
    if (value > 2147483648) break;
    i++;
  }
  value *= sign;
  return Math.max(-2147483648, Math.min(2147483647, value)) + 0;
}
//### zigzag-rows
function zigzagRead(text, rows) {
  if (rows === 1 || rows >= text.length) return text;
  const lines = Array.from({ length: rows }, () => []);
  let row = 0, step = 1;
  for (const ch of text) {
    lines[row].push(ch);
    if (row === 0) step = 1;
    else if (row === rows - 1) step = -1;
    row += step;
  }
  return lines.map((line) => line.join("")).join("");
}
//### one-edit-apart
function oneEditApart(first, second) {
  let a = first, b = second;
  if (a.length > b.length) { const t = a; a = b; b = t; }
  if (b.length - a.length > 1) return false;
  let i = 0;
  while (i < a.length && a[i] === b[i]) i++;
  if (i === a.length) return b.length - a.length === 1;
  if (a.length === b.length) return a.slice(i + 1) === b.slice(i + 1);
  return a.slice(i) === b.slice(i + 1);
}
//### multiply-digit-strings
function multiplyDigits(first, second) {
  if (first === "0" || second === "0") return "0";
  const result = new Array(first.length + second.length).fill(0);
  for (let i = first.length - 1; i >= 0; i--) {
    for (let j = second.length - 1; j >= 0; j--) {
      const total = (first.charCodeAt(i) - 48) * (second.charCodeAt(j) - 48) + result[i + j + 1];
      result[i + j + 1] = total % 10;
      result[i + j] += Math.floor(total / 10);
    }
  }
  let start = 0;
  while (start < result.length - 1 && result[start] === 0) start++;
  return result.slice(start).join("");
}
//### pattern-occurrences
function countOccurrences(text, pattern) {
  const m = pattern.length;
  const fail = new Array(m).fill(0);
  let k = 0;
  for (let i = 1; i < m; i++) {
    while (k > 0 && pattern[i] !== pattern[k]) k = fail[k - 1];
    if (pattern[i] === pattern[k]) k++;
    fail[i] = k;
  }
  let count = 0;
  k = 0;
  for (const ch of text) {
    while (k > 0 && ch !== pattern[k]) k = fail[k - 1];
    if (ch === pattern[k]) k++;
    if (k === m) { count++; k = fail[k - 1]; }
  }
  return count;
}
