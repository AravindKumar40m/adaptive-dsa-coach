//### daily-stock-span
vector<int> stock_spans(vector<int>& prices) {
    vector<pair<int,int>> stack;
    vector<int> out;
    for (int p : prices) {
        int span = 1;
        while (!stack.empty() && stack.back().first <= p) { span += stack.back().second; stack.pop_back(); }
        stack.push_back({p, span});
        out.push_back(span);
    }
    return out;
}
//### next-greater-circular
vector<int> next_greater_circular(vector<int>& numbers) {
    int n = (int) numbers.size();
    vector<int> out(n, -1), stack;
    for (int i = 0; i < 2 * n; i++) {
        int x = numbers[i % n];
        while (!stack.empty() && numbers[stack.back()] < x) { out[stack.back()] = x; stack.pop_back(); }
        if (i < n) stack.push_back(i);
    }
    return out;
}
//### simplify-path
string simplify_path(string path) {
    vector<string> stack;
    size_t i = 0;
    while (i < path.size()) {
        size_t j = path.find('/', i);
        if (j == string::npos) j = path.size();
        string part = path.substr(i, j - i);
        if (!part.empty() && part != ".") {
            if (part == "..") { if (!stack.empty()) stack.pop_back(); }
            else stack.push_back(part);
        }
        i = j + 1;
    }
    string out;
    for (auto& p : stack) out += "/" + p;
    return out.empty() ? "/" : out;
}
//### min-stack-queries
vector<int> min_stack_answers(vector<int>& ops) {
    vector<int> mins, out;
    for (int op : ops) {
        if (op > 0) {
            mins.push_back(mins.empty() ? op : min(op, mins.back()));
        } else if (op == 0) {
            if (!mins.empty()) mins.pop_back();
        } else {
            out.push_back(mins.empty() ? 0 : mins.back());
        }
    }
    return out;
}
//### longest-valid-brackets
int longest_valid_brackets(string text) {
    vector<int> stack;
    stack.push_back(-1);
    int best = 0;
    for (int i = 0; i < (int) text.size(); i++) {
        if (text[i] == '(') stack.push_back(i);
        else {
            stack.pop_back();
            if (stack.empty()) stack.push_back(i);
            else best = max(best, i - stack.back());
        }
    }
    return best;
}
//### calculator-with-brackets
int calculate_expression(string expression) {
    long long result = 0, number = 0;
    int sign = 1;
    vector<long long> stack;
    for (char ch : expression) {
        if (ch >= '0' && ch <= '9') number = number * 10 + (ch - '0');
        else if (ch == '+' || ch == '-') { result += sign * number; number = 0; sign = ch == '+' ? 1 : -1; }
        else if (ch == '(') { stack.push_back(result); stack.push_back(sign); result = 0; sign = 1; }
        else if (ch == ')') {
            result += sign * number;
            number = 0;
            long long outer_sign = stack.back(); stack.pop_back();
            long long outer_result = stack.back(); stack.pop_back();
            result = outer_result + outer_sign * result;
        }
    }
    return (int) (result + sign * number);
}
//### dedupe-sorted-list
ListNode* remove_duplicates_sorted(ListNode* head) {
    ListNode* cur = head;
    while (cur != nullptr && cur->next != nullptr) {
        if (cur->val == cur->next->val) cur->next = cur->next->next;
        else cur = cur->next;
    }
    return head;
}
//### partition-list-around
ListNode* partition_list(ListNode* head, int pivot) {
    ListNode less_head(0), more_head(0);
    ListNode* less = &less_head;
    ListNode* more = &more_head;
    while (head != nullptr) {
        if (head->val < pivot) { less->next = head; less = head; }
        else { more->next = head; more = head; }
        head = head->next;
    }
    more->next = nullptr;
    less->next = more_head.next;
    return less_head.next;
}
//### rotate-list-right
ListNode* rotate_list(ListNode* head, int k) {
    if (head == nullptr || head->next == nullptr) return head;
    int length = 1;
    ListNode* tail = head;
    while (tail->next != nullptr) { tail = tail->next; length++; }
    k %= length;
    if (k == 0) return head;
    tail->next = head;
    ListNode* new_tail = head;
    for (int i = 0; i < length - k - 1; i++) new_tail = new_tail->next;
    ListNode* new_head = new_tail->next;
    new_tail->next = nullptr;
    return new_head;
}
//### swap-adjacent-nodes
ListNode* swap_pairs(ListNode* head) {
    ListNode dummy(0);
    dummy.next = head;
    ListNode* prev = &dummy;
    while (prev->next != nullptr && prev->next->next != nullptr) {
        ListNode* a = prev->next;
        ListNode* b = a->next;
        a->next = b->next;
        b->next = a;
        prev->next = b;
        prev = a;
    }
    return dummy.next;
}
