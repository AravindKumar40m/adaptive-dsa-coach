//### remove-all-duplicate-values
function removeAllDuplicates(head) {
  const dummy = new ListNode(0, head);
  let prev = dummy, cur = head;
  while (cur !== null) {
    if (cur.next !== null && cur.next.val === cur.val) {
      const v = cur.val;
      while (cur !== null && cur.val === v) cur = cur.next;
      prev.next = cur;
    } else {
      prev = cur;
      cur = cur.next;
    }
  }
  return dummy.next;
}
//### double-digit-list
function doubleDigitList(head) {
  const digits = [];
  for (let node = head; node !== null; node = node.next) digits.push(node.val);
  let carry = 0;
  for (let i = digits.length - 1; i >= 0; i--) {
    const v = digits[i] * 2 + carry;
    digits[i] = v % 10;
    carry = Math.floor(v / 10);
  }
  if (carry > 0) digits.unshift(carry);
  const dummy = new ListNode(0, null);
  let tail = dummy;
  for (const d of digits) { tail.next = new ListNode(d, null); tail = tail.next; }
  return dummy.next;
}
//### min-depth
function treeMinDepth(root) {
  if (root === null) return 0;
  const queue = [[root, 1]];
  for (let head = 0; head < queue.length; head++) {
    const [node, level] = queue[head];
    if (node.left === null && node.right === null) return level;
    if (node.left !== null) queue.push([node.left, level + 1]);
    if (node.right !== null) queue.push([node.right, level + 1]);
  }
  return 0;
}
//### sum-left-leaves
function sumLeftLeaves(root) {
  function walk(node) {
    if (node === null) return 0;
    let total = 0;
    if (node.left !== null && node.left.left === null && node.left.right === null) total += node.left.val;
    else total += walk(node.left);
    return total + walk(node.right);
  }
  return walk(root);
}
//### is-subtree
function isSubtree(root, sub) {
  function text(node) {
    const parts = [], stack = [node];
    while (stack.length > 0) {
      const cur = stack.pop();
      if (cur === null) parts.push("#");
      else { parts.push("^" + cur.val); stack.push(cur.right); stack.push(cur.left); }
    }
    return "," + parts.join(",") + ",";
  }
  return text(root).includes(text(sub));
}
//### longest-univalue-path
function longestUnivaluePath(root) {
  let best = 0;
  function down(node) {
    if (node === null) return 0;
    const left = down(node.left), right = down(node.right);
    const leftArm = node.left !== null && node.left.val === node.val ? left + 1 : 0;
    const rightArm = node.right !== null && node.right.val === node.val ? right + 1 : 0;
    best = Math.max(best, leftArm + rightArm);
    return Math.max(leftArm, rightArm);
  }
  down(root);
  return best;
}
//### flatten-to-right-chain
function flattenToChain(root) {
  let cur = root;
  while (cur !== null) {
    if (cur.left !== null) {
      let last = cur.left;
      while (last.right !== null) last = last.right;
      last.right = cur.right;
      cur.right = cur.left;
      cur.left = null;
    }
    cur = cur.right;
  }
  return root;
}
//### nodes-at-distance-k
function countNodesAtDistance(root, target, k) {
  let found = 0;
  function collect(node, depth) {
    if (node === null) return 0;
    if (depth === 0) return 1;
    return collect(node.left, depth - 1) + collect(node.right, depth - 1);
  }
  function go(node) {
    if (node === null) return -1;
    if (node.val === target) { found += collect(node, k); return 0; }
    const left = go(node.left);
    if (left !== -1) {
      const dist = left + 1;
      if (dist === k) found += 1;
      else if (dist < k) found += collect(node.right, k - dist - 1);
      return dist;
    }
    const right = go(node.right);
    if (right !== -1) {
      const dist = right + 1;
      if (dist === k) found += 1;
      else if (dist < k) found += collect(node.left, k - dist - 1);
      return dist;
    }
    return -1;
  }
  go(root);
  return found;
}
//### hanoi-move-count
function hanoiMoveCount(n) {
  return Math.pow(2, n) - 1;
}
//### strobogrammatic-count
function strobogrammaticCount(n) {
  if (n === 1) return 3;
  const pairs = Math.floor(n / 2);
  let result = n % 2 === 1 ? 3 : 1;
  for (let i = 0; i < pairs; i++) result *= i === 0 ? 4 : 5;
  return result;
}
