//### digital-root-recursive
class Solution {
    public int digitalRoot(int number) {
        return number == 0 ? 0 : 1 + (number - 1) % 9;
    }
}
//### count-good-numbers
class Solution {
    private static final long MOD = 1000000007L;

    private long power(long base, long exponent) {
        if (exponent == 0) return 1;
        long half = power(base, exponent / 2);
        long result = half * half % MOD;
        return exponent % 2 == 1 ? result * base % MOD : result;
    }

    public int countGoodNumbers(int length) {
        long n = length;
        return (int) (power(5, (n + 1) / 2) * power(4, n / 2) % MOD);
    }
}
//### ackermann-value
class Solution {
    public int ackermann(int m, int n) {
        if (m == 0) return n + 1;
        if (m == 1) return n + 2;
        if (m == 2) return 2 * n + 3;
        return (1 << (n + 3)) - 3;
    }
}
//### restore-ip-count
class Solution {
    private int go(String digits, int pos, int parts) {
        if (parts == 4) return pos == digits.length() ? 1 : 0;
        int total = 0;
        for (int size = 1; size <= 3; size++) {
            if (pos + size > digits.length()) break;
            String part = digits.substring(pos, pos + size);
            if ((size > 1 && part.charAt(0) == '0') || Integer.parseInt(part) > 255) continue;
            total += go(digits, pos + size, parts + 1);
        }
        return total;
    }

    public int countIpAddresses(String digits) {
        return go(digits, 0, 0);
    }
}
//### combination-sum-distinct-count
class Solution {
    private int[] a;

    private int go(int start, int left) {
        if (left == 0) return 1;
        int total = 0;
        for (int i = start; i < a.length; i++) {
            if (i > start && a[i] == a[i - 1]) continue;
            if (a[i] > left) break;
            total += go(i + 1, left - a[i]);
        }
        return total;
    }

    public int countCombinations(int[] numbers, int target) {
        a = numbers.clone();
        Arrays.sort(a);
        return go(0, target);
    }
}
//### distinct-subsets-count
class Solution {
    public int countDistinctSubsets(int[] numbers) {
        Map<Integer, Integer> counts = new HashMap<>();
        for (int x : numbers) counts.merge(x, 1, Integer::sum);
        long result = 1;
        for (int c : counts.values()) result *= c + 1;
        return (int) result;
    }
}
//### close-neighbour-orders
class Solution {
    private int[] numbers;
    private boolean[] used;
    private int gap;

    private int go(int placed, int last) {
        if (placed == numbers.length) return 1;
        int total = 0;
        for (int i = 0; i < numbers.length; i++) {
            if (!used[i] && (placed == 0 || Math.abs(numbers[i] - last) <= gap)) {
                used[i] = true;
                total += go(placed + 1, numbers[i]);
                used[i] = false;
            }
        }
        return total;
    }

    public int countSmoothOrders(int[] numbers, int gap) {
        this.numbers = numbers;
        this.gap = gap;
        this.used = new boolean[numbers.length];
        return go(0, 0);
    }
}
//### target-expression-count
class Solution {
    private String digits;
    private long target;

    private int go(int pos, long value, long last) {
        int n = digits.length();
        if (pos == n) return value == target ? 1 : 0;
        int total = 0;
        for (int end = pos + 1; end <= n; end++) {
            if (end - pos > 1 && digits.charAt(pos) == '0') break;
            long num = Long.parseLong(digits.substring(pos, end));
            if (pos == 0) total += go(end, num, num);
            else {
                total += go(end, value + num, num);
                total += go(end, value - num, -num);
                total += go(end, value - last + last * num, last * num);
            }
        }
        return total;
    }

    public int countExpressions(String digits, int target) {
        this.digits = digits;
        this.target = target;
        return go(0, 0, 0);
    }
}
//### fair-cookie-split
class Solution {
    private int[] items;
    private int[] totals;
    private int best;

    private void go(int i) {
        if (i == items.length) {
            int worst = 0;
            for (int t : totals) worst = Math.max(worst, t);
            best = Math.min(best, worst);
            return;
        }
        Set<Integer> seen = new HashSet<>();
        for (int c = 0; c < totals.length; c++) {
            if (!seen.add(totals[c])) continue;
            if (totals[c] + items[i] >= best) continue;
            totals[c] += items[i];
            go(i + 1);
            totals[c] -= items[i];
        }
    }

    public int fairCookieSplit(int[] bags, int children) {
        items = bags.clone();
        Arrays.sort(items);
        for (int a = 0, b = items.length - 1; a < b; a++, b--) { int t = items[a]; items[a] = items[b]; items[b] = t; }
        totals = new int[children];
        best = 1;
        for (int x : items) best += x;
        go(0);
        return best;
    }
}
//### matrix-search-sorted-rows
class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int cols = matrix[0].length;
        int lo = 0, hi = matrix.length * cols - 1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            int value = matrix[mid / cols][mid % cols];
            if (value == target) return true;
            if (value < target) lo = mid + 1; else hi = mid - 1;
        }
        return false;
    }
}
