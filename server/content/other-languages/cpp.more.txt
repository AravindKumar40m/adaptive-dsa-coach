//### second-highest
int second_highest(vector<int>& scores) {
    int first = -1, second = -1;
    for (int x : scores) {
        if (x > first) { second = first; first = x; }
        else if (x < first && x > second) second = x;
    }
    return second;
}
//### best-streak-sum
int best_streak_sum(vector<int>& daily) {
    int current = daily[0], best = daily[0];
    for (size_t i = 1; i < daily.size(); i++) {
        current = max(daily[i], current + daily[i]);
        best = max(best, current);
    }
    return best;
}
//### first-unique-char
int first_unique_index(string text) {
    int count[26] = {0};
    for (char ch : text) count[ch - 'a']++;
    for (size_t i = 0; i < text.size(); i++) {
        if (count[text[i] - 'a'] == 1) return (int) i;
    }
    return -1;
}
//### subarray-sum-count
int count_windows(vector<int>& readings, int target) {
    unordered_map<int, int> seen;
    seen[0] = 1;
    int running = 0, count = 0;
    for (int x : readings) {
        running += x;
        auto it = seen.find(running - target);
        if (it != seen.end()) count += it->second;
        seen[running]++;
    }
    return count;
}
//### widest-fence
int max_banner_area(vector<int>& heights) {
    int i = 0, j = (int) heights.size() - 1, best = 0;
    while (i < j) {
        best = max(best, min(heights[i], heights[j]) * (j - i));
        if (heights[i] < heights[j]) i++; else j--;
    }
    return best;
}
//### peak-overlap
int max_overlap(vector<vector<int>>& talks) {
    vector<int> starts, ends;
    for (auto& t : talks) { starts.push_back(t[0]); ends.push_back(t[1]); }
    sort(starts.begin(), starts.end());
    sort(ends.begin(), ends.end());
    size_t i = 0, j = 0;
    int running = 0, best = 0;
    while (i < starts.size()) {
        if (starts[i] < ends[j]) { running++; i++; best = max(best, running); }
        else { running--; j++; }
    }
    return best;
}
//### longest-unique-run
int longest_unique_run(string text) {
    int last[26];
    for (int c = 0; c < 26; c++) last[c] = -1;
    int start = 0, best = 0;
    for (int i = 0; i < (int) text.size(); i++) {
        int c = text[i] - 'a';
        if (last[c] >= start) start = last[c] + 1;
        last[c] = i;
        best = max(best, i - start + 1);
    }
    return best;
}
//### postfix-eval
int eval_postfix(string expr) {
    vector<int> stack;
    istringstream in(expr);
    string token;
    while (in >> token) {
        if (token == "+" || token == "-" || token == "*") {
            int b = stack.back(); stack.pop_back();
            int a = stack.back(); stack.pop_back();
            stack.push_back(token == "+" ? a + b : token == "-" ? a - b : a * b);
        } else {
            stack.push_back(stoi(token));
        }
    }
    return stack.back();
}
//### fair-capacity
int min_capacity(vector<int>& packages, int days) {
    auto days_needed = [&](int cap) {
        int d = 1, load = 0;
        for (int w : packages) {
            if (load + w > cap) { d++; load = 0; }
            load += w;
        }
        return d;
    };
    int lo = *max_element(packages.begin(), packages.end());
    int hi = accumulate(packages.begin(), packages.end(), 0);
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (days_needed(mid) <= days) hi = mid; else lo = mid + 1;
    }
    return lo;
}
//### subset-sum-count
static int subset_go(const vector<int>& weights, size_t i, int remaining) {
    if (remaining < 0) return 0;
    if (i == weights.size()) return remaining == 0 ? 1 : 0;
    return subset_go(weights, i + 1, remaining - weights[i]) + subset_go(weights, i + 1, remaining);
}
int count_selections(vector<int>& weights, int target) {
    return subset_go(weights, 0, target);
}
//### skip-neighbours
int max_prizes(vector<int>& prizes) {
    int prev2 = 0, prev1 = 0;
    for (int p : prizes) {
        int current = max(prev1, prev2 + p);
        prev2 = prev1;
        prev1 = current;
    }
    return prev1;
}
//### min-coins
int fewest_coins(vector<int>& coins, int amount) {
    int inf = amount + 1;
    vector<int> best(amount + 1, inf);
    best[0] = 0;
    for (int a = 1; a <= amount; a++) {
        for (int c : coins) {
            if (c <= a && best[a - c] + 1 < best[a]) best[a] = best[a - c] + 1;
        }
    }
    return best[amount] == inf ? -1 : best[amount];
}
//### grid-paths
int count_paths(vector<vector<int>>& grid) {
    int rows = (int) grid.size(), cols = (int) grid[0].size();
    vector<int> row(cols, 0);
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == 1) row[c] = 0;
            else if (r == 0 && c == 0) row[c] = 1;
            else if (c > 0) row[c] += row[c - 1];
        }
    }
    return row[cols - 1];
}
//### cookie-match
int max_happy(vector<int>& needs, vector<int>& sizes) {
    vector<int> n = needs, s = sizes;
    sort(n.begin(), n.end());
    sort(s.begin(), s.end());
    size_t i = 0;
    for (int c : s) {
        if (i < n.size() && c >= n[i]) i++;
    }
    return (int) i;
}
//### max-non-overlap
int max_jobs(vector<vector<int>>& jobs) {
    vector<vector<int>> sorted = jobs;
    sort(sorted.begin(), sorted.end(), [](const vector<int>& a, const vector<int>& b) { return a[1] < b[1]; });
    int last_end = -1, count = 0;
    for (auto& job : sorted) {
        if (job[0] >= last_end) { count++; last_end = job[1]; }
    }
    return count;
}
//### kth-largest
int kth_largest(vector<int>& scores, int k) {
    priority_queue<int, vector<int>, greater<int>> heap;
    for (int x : scores) {
        heap.push(x);
        if ((int) heap.size() > k) heap.pop();
    }
    return heap.top();
}
//### rope-join
int min_join_cost(vector<int>& ropes) {
    priority_queue<int, vector<int>, greater<int>> heap(ropes.begin(), ropes.end());
    int total = 0;
    while (heap.size() > 1) {
        int a = heap.top(); heap.pop();
        int b = heap.top(); heap.pop();
        total += a + b;
        heap.push(a + b);
    }
    return total;
}
//### connected-groups
static int groups_find(vector<int>& parent, int x) {
    while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }
    return x;
}
int count_groups(int n, vector<vector<int>>& friendships) {
    vector<int> parent(n);
    for (int i = 0; i < n; i++) parent[i] = i;
    int groups = n;
    for (auto& f : friendships) {
        int ra = groups_find(parent, f[0]), rb = groups_find(parent, f[1]);
        if (ra != rb) { parent[ra] = rb; groups--; }
    }
    return groups;
}
//### grid-islands
int count_islands(vector<vector<int>>& grid) {
    int rows = (int) grid.size(), cols = (int) grid[0].size();
    vector<vector<bool>> seen(rows, vector<bool>(cols, false));
    int dr[4] = {1, -1, 0, 0}, dc[4] = {0, 0, 1, -1};
    int islands = 0;
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == 1 && !seen[r][c]) {
                islands++;
                seen[r][c] = true;
                vector<pair<int, int>> stack = {{r, c}};
                while (!stack.empty()) {
                    auto [cr, cc] = stack.back();
                    stack.pop_back();
                    for (int d = 0; d < 4; d++) {
                        int nr = cr + dr[d], nc = cc + dc[d];
                        if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] == 1 && !seen[nr][nc]) {
                            seen[nr][nc] = true;
                            stack.push_back({nr, nc});
                        }
                    }
                }
            }
        }
    }
    return islands;
}
//### shortest-grid-path
int shortest_path(vector<vector<int>>& grid) {
    int rows = (int) grid.size(), cols = (int) grid[0].size();
    if (grid[0][0] == 1 || grid[rows - 1][cols - 1] == 1) return -1;
    vector<vector<int>> dist(rows, vector<int>(cols, -1));
    dist[0][0] = 0;
    int dr[4] = {1, -1, 0, 0}, dc[4] = {0, 0, 1, -1};
    queue<pair<int, int>> q;
    q.push({0, 0});
    while (!q.empty()) {
        auto [r, c] = q.front();
        q.pop();
        if (r == rows - 1 && c == cols - 1) return dist[r][c];
        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] == 0 && dist[nr][nc] == -1) {
                dist[nr][nc] = dist[r][c] + 1;
                q.push({nr, nc});
            }
        }
    }
    return -1;
}
