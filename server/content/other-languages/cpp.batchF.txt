//### integer-sqrt
int integer_sqrt(int n) {
    long long lo = 0, hi = min(n, 46341);
    while (lo < hi) {
        long long mid = (lo + hi + 1) / 2;
        if (mid * mid <= n) lo = mid; else hi = mid - 1;
    }
    return (int) lo;
}
//### search-sorted
int find_index(vector<int>& nums, int target) {
    int lo = 0, hi = (int) nums.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] == target) return mid;
        if (nums[mid] < target) lo = mid + 1; else hi = mid - 1;
    }
    return -1;
}
//### search-rotated
int search_rotated(vector<int>& nums, int target) {
    int lo = 0, hi = (int) nums.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] == target) return mid;
        if (nums[lo] <= nums[mid]) {
            if (nums[lo] <= target && target < nums[mid]) hi = mid - 1; else lo = mid + 1;
        } else {
            if (nums[mid] < target && target <= nums[hi]) lo = mid + 1; else hi = mid - 1;
        }
    }
    return -1;
}
//### mountain-peak
int peak_index(vector<int>& heights) {
    int lo = 0, hi = (int) heights.size() - 1;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (heights[mid] < heights[mid + 1]) lo = mid + 1; else hi = mid;
    }
    return lo;
}
//### min-rotated
int min_in_rotated(vector<int>& nums) {
    int lo = 0, hi = (int) nums.size() - 1;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] > nums[hi]) lo = mid + 1; else hi = mid;
    }
    return nums[lo];
}
//### median-two-sorted
int median_times_two(vector<int>& first, vector<int>& second) {
    vector<int>& a = first.size() <= second.size() ? first : second;
    vector<int>& b = first.size() <= second.size() ? second : first;
    int m = (int) a.size(), n = (int) b.size(), half = (m + n + 1) / 2;
    int lo = 0, hi = m;
    const long long NEG = LLONG_MIN / 4, POS = LLONG_MAX / 4;
    while (lo <= hi) {
        int i = lo + (hi - lo) / 2, j = half - i;
        long long a_left = i > 0 ? a[i - 1] : NEG, a_right = i < m ? a[i] : POS;
        long long b_left = j > 0 ? b[j - 1] : NEG, b_right = j < n ? b[j] : POS;
        if (a_left <= b_right && b_left <= a_right) {
            if ((m + n) % 2 == 1) return (int) (2 * max(a_left, b_left));
            return (int) (max(a_left, b_left) + min(a_right, b_right));
        }
        if (a_left > b_right) hi = i - 1; else lo = i + 1;
    }
    return 0;
}
//### max-profit-trades
int max_profit(vector<int>& prices) {
    int profit = 0;
    for (size_t i = 1; i < prices.size(); i++) if (prices[i] > prices[i - 1]) profit += prices[i] - prices[i - 1];
    return profit;
}
//### lemonade-change
bool can_give_change(vector<int>& bills) {
    int fives = 0, tens = 0;
    for (int bill : bills) {
        if (bill == 5) fives++;
        else if (bill == 10) {
            if (fives == 0) return false;
            fives--; tens++;
        } else if (tens > 0 && fives > 0) { tens--; fives--; }
        else if (fives >= 3) fives -= 3;
        else return false;
    }
    return true;
}
//### jump-reach
bool can_reach_end(vector<int>& jumps) {
    long long farthest = 0;
    for (int i = 0; i < (int) jumps.size(); i++) {
        if (i > farthest) return false;
        farthest = max(farthest, (long long) i + jumps[i]);
    }
    return true;
}
//### min-jumps
int min_jumps(vector<int>& jumps) {
    int used = 0, range_end = 0, farthest = 0;
    for (int i = 0; i + 1 < (int) jumps.size(); i++) {
        farthest = max(farthest, i + jumps[i]);
        if (i == range_end) { used++; range_end = farthest; }
    }
    return used;
}
//### candy-distribution
int min_candies(vector<int>& ratings) {
    int n = (int) ratings.size();
    vector<int> candies(n, 1);
    for (int i = 1; i < n; i++) if (ratings[i] > ratings[i - 1]) candies[i] = candies[i - 1] + 1;
    for (int i = n - 2; i >= 0; i--) if (ratings[i] > ratings[i + 1]) candies[i] = max(candies[i], candies[i + 1] + 1);
    long long total = 0;
    for (int c : candies) total += c;
    return (int) total;
}
