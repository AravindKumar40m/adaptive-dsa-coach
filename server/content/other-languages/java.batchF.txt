//### integer-sqrt
class Solution {
    public int integerSqrt(int n) {
        long lo = 0, hi = Math.min(n, 46341);
        while (lo < hi) {
            long mid = (lo + hi + 1) / 2;
            if (mid * mid <= n) lo = mid; else hi = mid - 1;
        }
        return (int) lo;
    }
}
//### search-sorted
class Solution {
    public int findIndex(int[] nums, int target) {
        int lo = 0, hi = nums.length - 1;
        while (lo <= hi) {
            int mid = (lo + hi) >>> 1;
            if (nums[mid] == target) return mid;
            if (nums[mid] < target) lo = mid + 1; else hi = mid - 1;
        }
        return -1;
    }
}
//### search-rotated
class Solution {
    public int searchRotated(int[] nums, int target) {
        int lo = 0, hi = nums.length - 1;
        while (lo <= hi) {
            int mid = (lo + hi) >>> 1;
            if (nums[mid] == target) return mid;
            if (nums[lo] <= nums[mid]) {
                if (nums[lo] <= target && target < nums[mid]) hi = mid - 1; else lo = mid + 1;
            } else {
                if (nums[mid] < target && target <= nums[hi]) lo = mid + 1; else hi = mid - 1;
            }
        }
        return -1;
    }
}
//### mountain-peak
class Solution {
    public int peakIndex(int[] heights) {
        int lo = 0, hi = heights.length - 1;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (heights[mid] < heights[mid + 1]) lo = mid + 1; else hi = mid;
        }
        return lo;
    }
}
//### min-rotated
class Solution {
    public int minInRotated(int[] nums) {
        int lo = 0, hi = nums.length - 1;
        while (lo < hi) {
            int mid = (lo + hi) >>> 1;
            if (nums[mid] > nums[hi]) lo = mid + 1; else hi = mid;
        }
        return nums[lo];
    }
}
//### median-two-sorted
class Solution {
    public int medianTimesTwo(int[] first, int[] second) {
        int[] a = first.length <= second.length ? first : second;
        int[] b = first.length <= second.length ? second : first;
        int m = a.length, n = b.length, half = (m + n + 1) / 2;
        int lo = 0, hi = m;
        while (lo <= hi) {
            int i = (lo + hi) >>> 1, j = half - i;
            long aLeft = i > 0 ? a[i - 1] : Long.MIN_VALUE, aRight = i < m ? a[i] : Long.MAX_VALUE;
            long bLeft = j > 0 ? b[j - 1] : Long.MIN_VALUE, bRight = j < n ? b[j] : Long.MAX_VALUE;
            if (aLeft <= bRight && bLeft <= aRight) {
                if ((m + n) % 2 == 1) return (int) (2 * Math.max(aLeft, bLeft));
                return (int) (Math.max(aLeft, bLeft) + Math.min(aRight, bRight));
            }
            if (aLeft > bRight) hi = i - 1; else lo = i + 1;
        }
        return 0;
    }
}
//### max-profit-trades
class Solution {
    public int maxProfit(int[] prices) {
        int profit = 0;
        for (int i = 1; i < prices.length; i++) if (prices[i] > prices[i - 1]) profit += prices[i] - prices[i - 1];
        return profit;
    }
}
//### lemonade-change
class Solution {
    public boolean canGiveChange(int[] bills) {
        int fives = 0, tens = 0;
        for (int bill : bills) {
            if (bill == 5) fives++;
            else if (bill == 10) {
                if (fives == 0) return false;
                fives--; tens++;
            } else if (tens > 0 && fives > 0) { tens--; fives--; }
            else if (fives >= 3) fives -= 3;
            else return false;
        }
        return true;
    }
}
//### jump-reach
class Solution {
    public boolean canReachEnd(int[] jumps) {
        long farthest = 0;
        for (int i = 0; i < jumps.length; i++) {
            if (i > farthest) return false;
            farthest = Math.max(farthest, (long) i + jumps[i]);
        }
        return true;
    }
}
//### min-jumps
class Solution {
    public int minJumps(int[] jumps) {
        int used = 0, rangeEnd = 0, farthest = 0;
        for (int i = 0; i < jumps.length - 1; i++) {
            farthest = Math.max(farthest, i + jumps[i]);
            if (i == rangeEnd) { used++; rangeEnd = farthest; }
        }
        return used;
    }
}
//### candy-distribution
class Solution {
    public int minCandies(int[] ratings) {
        int n = ratings.length;
        int[] candies = new int[n];
        Arrays.fill(candies, 1);
        for (int i = 1; i < n; i++) if (ratings[i] > ratings[i - 1]) candies[i] = candies[i - 1] + 1;
        for (int i = n - 2; i >= 0; i--) if (ratings[i] > ratings[i + 1]) candies[i] = Math.max(candies[i], candies[i + 1] + 1);
        long total = 0;
        for (int c : candies) total += c;
        return (int) total;
    }
}
