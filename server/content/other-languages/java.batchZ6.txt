//### tribonacci-number
class Solution {
    public int tribonacciNumber(int n) {
        if (n == 0) return 0;
        long a = 0, b = 1, c = 1;
        for (int i = 3; i <= n; i++) { long next = a + b + c; a = b; b = c; c = next; }
        return (int) c;
    }
}
//### unique-digit-numbers-count
class Solution {
    public int uniqueDigitCount(int n) {
        long total = 1;
        for (int length = 1; length <= n; length++) {
            long ways = 9;
            int available = 9;
            for (int i = 1; i < length; i++) { ways *= available; available--; }
            total += ways;
        }
        return (int) total;
    }
}
//### kth-lexicographic-number
class Solution {
    private long subtree(long prefix, long n) {
        long first = prefix, last = prefix, total = 0;
        while (first <= n) {
            total += Math.min(n + 1, last + 1) - first;
            first *= 10;
            last = last * 10 + 9;
        }
        return total;
    }

    public int kthLexicographic(int n, int k) {
        long current = 1;
        long remaining = k - 1;
        while (remaining > 0) {
            long size = subtree(current, n);
            if (size <= remaining) { remaining -= size; current += 1; }
            else { current *= 10; remaining -= 1; }
        }
        return (int) current;
    }
}
//### grid-path-spell
class Solution {
    private int[][] grid;
    private int[] word;
    private boolean[][] used;

    private boolean go(int r, int c, int i) {
        if (i == word.length - 1) return true;
        used[r][c] = true;
        int[] dr = {1, -1, 0, 0}, dc = {0, 0, 1, -1};
        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr >= 0 && nr < grid.length && nc >= 0 && nc < grid[0].length && !used[nr][nc] && grid[nr][nc] == word[i + 1]) {
                if (go(nr, nc, i + 1)) { used[r][c] = false; return true; }
            }
        }
        used[r][c] = false;
        return false;
    }

    public boolean gridHasPath(int[][] grid, int[] word) {
        if (word.length == 0) return true;
        this.grid = grid;
        this.word = word;
        this.used = new boolean[grid.length][grid[0].length];
        for (int r = 0; r < grid.length; r++)
            for (int c = 0; c < grid[0].length; c++)
                if (grid[r][c] == word[0] && go(r, c, 0)) return true;
        return false;
    }
}
//### combination-sum-three
class Solution {
    private int go(int start, int left, int remaining) {
        if (left == 0) return remaining == 0 ? 1 : 0;
        int total = 0;
        for (int d = start; d <= 9; d++) {
            if (d > remaining) break;
            total += go(d + 1, left - 1, remaining - d);
        }
        return total;
    }

    public int countCombinationThree(int k, int n) {
        return go(1, k, n);
    }
}
//### beautiful-arrangements-count
class Solution {
    public int beautifulArrangements(int n) {
        int[] dp = new int[1 << n];
        dp[0] = 1;
        for (int mask = 0; mask < (1 << n); mask++) {
            if (dp[mask] == 0) continue;
            int pos = Integer.bitCount(mask) + 1;
            for (int num = 1; num <= n; num++) {
                int bit = 1 << (num - 1);
                if ((mask & bit) == 0 && (num % pos == 0 || pos % num == 0)) dp[mask | bit] += dp[mask];
            }
        }
        return dp[(1 << n) - 1];
    }
}
//### consecutive-descending-split
class Solution {
    private String digits;

    private boolean go(int start, long previous) {
        int n = digits.length();
        if (start == n) return true;
        long need = previous - 1;
        for (int end = start + 1; end <= n; end++) {
            long v = Long.parseLong(digits.substring(start, end));
            if (v == need && go(end, v)) return true;
            if (v > need) break;
        }
        return false;
    }

    public boolean isConsecutiveDescending(String digits) {
        this.digits = digits;
        for (int end = 1; end < digits.length(); end++) {
            if (go(end, Long.parseLong(digits.substring(0, end)))) return true;
        }
        return false;
    }
}
//### max-unique-split
class Solution {
    private String text;
    private int best;
    private Set<String> used;

    private void go(int start, int count) {
        int n = text.length();
        if (count + (n - start) <= best) return;
        if (start == n) { best = Math.max(best, count); return; }
        for (int end = start + 1; end <= n; end++) {
            String piece = text.substring(start, end);
            if (!used.contains(piece)) {
                used.add(piece);
                go(end, count + 1);
                used.remove(piece);
            }
        }
    }

    public int maxUniqueSplit(String text) {
        this.text = text;
        this.best = 0;
        this.used = new HashSet<>();
        go(0, 0);
        return best;
    }
}
//### shortest-tour-all-cities
class Solution {
    public int shortestTour(int[][] dist) {
        int n = dist.length, INF = 1000000000;
        int[][] dp = new int[1 << n][n];
        for (int[] row : dp) Arrays.fill(row, INF);
        dp[1][0] = 0;
        for (int mask = 1; mask < (1 << n); mask++) {
            if ((mask & 1) == 0) continue;
            for (int last = 0; last < n; last++) {
                if (dp[mask][last] >= INF) continue;
                for (int nxt = 0; nxt < n; nxt++) {
                    if ((mask & (1 << nxt)) != 0) continue;
                    int cost = dp[mask][last] + dist[last][nxt];
                    if (cost < dp[mask | (1 << nxt)][nxt]) dp[mask | (1 << nxt)][nxt] = cost;
                }
            }
        }
        int best = INF;
        for (int last = 0; last < n; last++) best = Math.min(best, dp[(1 << n) - 1][last]);
        return best;
    }
}
//### first-true-position
class Solution {
    public int firstTrueIndex(int[] flags) {
        int lo = 0, hi = flags.length;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (flags[mid] == 1) hi = mid; else lo = mid + 1;
        }
        return lo < flags.length ? lo : -1;
    }
}
