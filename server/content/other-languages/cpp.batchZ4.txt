//### valid-number-format
bool valid_number(string text) {
    int n = (int) text.size(), i = 0;
    if (i < n && (text[i] == '+' || text[i] == '-')) i++;
    int before = 0;
    while (i < n && text[i] >= '0' && text[i] <= '9') { i++; before++; }
    int after = 0;
    if (i < n && text[i] == '.') {
        i++;
        while (i < n && text[i] >= '0' && text[i] <= '9') { i++; after++; }
    }
    if (before + after == 0) return false;
    if (i < n && (text[i] == 'e' || text[i] == 'E')) {
        i++;
        if (i < n && (text[i] == '+' || text[i] == '-')) i++;
        int exponent = 0;
        while (i < n && text[i] >= '0' && text[i] <= '9') { i++; exponent++; }
        if (exponent == 0) return false;
    }
    return i == n;
}
//### smallest-distinct-subsequence
string smallest_distinct_subsequence(string text) {
    int last[26];
    for (int i = 0; i < 26; i++) last[i] = -1;
    for (int i = 0; i < (int) text.size(); i++) last[text[i] - 'a'] = i;
    string stack;
    bool in_stack[26] = {false};
    for (int i = 0; i < (int) text.size(); i++) {
        char c = text[i];
        if (in_stack[c - 'a']) continue;
        while (!stack.empty() && stack.back() > c && last[stack.back() - 'a'] > i) {
            in_stack[stack.back() - 'a'] = false;
            stack.pop_back();
        }
        stack += c;
        in_stack[c - 'a'] = true;
    }
    return stack;
}
//### remove-k-digits
string remove_k_digits(string num, int k) {
    string stack;
    for (char d : num) {
        while (k > 0 && !stack.empty() && stack.back() > d) { stack.pop_back(); k--; }
        stack += d;
    }
    while (k > 0) { stack.pop_back(); k--; }
    size_t start = 0;
    while (start < stack.size() && stack[start] == '0') start++;
    string result = stack.substr(start);
    return result.empty() ? "0" : result;
}
//### pattern-132-exists
bool has_132_pattern(vector<int>& numbers) {
    long long third = LLONG_MIN;
    vector<int> stack;
    for (int i = (int) numbers.size() - 1; i >= 0; i--) {
        int x = numbers[i];
        if (x < third) return true;
        while (!stack.empty() && stack.back() < x) { third = stack.back(); stack.pop_back(); }
        stack.push_back(x);
    }
    return false;
}
//### sum-subarray-minimums
int sum_subarray_mins(vector<int>& numbers) {
    const long long MOD = 1000000007LL;
    int n = (int) numbers.size();
    vector<int> left(n), right(n), stack;
    for (int i = 0; i < n; i++) {
        while (!stack.empty() && numbers[stack.back()] > numbers[i]) stack.pop_back();
        left[i] = i - (stack.empty() ? -1 : stack.back());
        stack.push_back(i);
    }
    stack.clear();
    for (int i = n - 1; i >= 0; i--) {
        while (!stack.empty() && numbers[stack.back()] >= numbers[i]) stack.pop_back();
        right[i] = (stack.empty() ? n : stack.back()) - i;
        stack.push_back(i);
    }
    long long total = 0;
    for (int i = 0; i < n; i++) total = (total + (long long) numbers[i] * left[i] % MOD * right[i]) % MOD;
    return (int) total;
}
//### reverse-in-brackets
string reverse_in_brackets(string text) {
    int n = (int) text.size();
    vector<int> partner(n, -1), opens;
    for (int i = 0; i < n; i++) {
        if (text[i] == '(') opens.push_back(i);
        else if (text[i] == ')') { int j = opens.back(); opens.pop_back(); partner[i] = j; partner[j] = i; }
    }
    string out;
    int i = 0, step = 1;
    while (i >= 0 && i < n) {
        if (text[i] == '(' || text[i] == ')') { i = partner[i]; step = -step; }
        else out += text[i];
        i += step;
    }
    return out;
}
//### car-fleet-count
int count_car_fleets(int target, vector<int>& position, vector<int>& speed) {
    int n = (int) position.size();
    vector<int> order(n);
    for (int i = 0; i < n; i++) order[i] = i;
    sort(order.begin(), order.end(), [&](int a, int b) { return position[a] > position[b]; });
    int fleets = 0;
    long long lead_distance = 0, lead_speed = 1;
    for (int i : order) {
        long long distance = target - position[i];
        if (fleets == 0 || distance * lead_speed > lead_distance * speed[i]) {
            fleets++;
            lead_distance = distance;
            lead_speed = speed[i];
        }
    }
    return fleets;
}
//### remove-list-elements
ListNode* remove_list_values(ListNode* head, int val) {
    ListNode dummy(0);
    dummy.next = head;
    ListNode* prev = &dummy;
    while (prev->next != nullptr) {
        if (prev->next->val == val) prev->next = prev->next->next;
        else prev = prev->next;
    }
    return dummy.next;
}
//### delete-middle-node
ListNode* delete_middle(ListNode* head) {
    if (head == nullptr || head->next == nullptr) return nullptr;
    ListNode* prev = nullptr;
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast != nullptr && fast->next != nullptr) {
        prev = slow;
        slow = slow->next;
        fast = fast->next->next;
    }
    prev->next = slow->next;
    return head;
}
//### reorder-list-alternating
ListNode* reorder_alternating(ListNode* head) {
    if (head == nullptr || head->next == nullptr) return head;
    ListNode* slow = head;
    ListNode* fast = head;
    while (fast->next != nullptr && fast->next->next != nullptr) { slow = slow->next; fast = fast->next->next; }
    ListNode* second = slow->next;
    slow->next = nullptr;
    ListNode* prev = nullptr;
    while (second != nullptr) { ListNode* nxt = second->next; second->next = prev; prev = second; second = nxt; }
    ListNode* first = head;
    second = prev;
    while (second != nullptr) {
        ListNode* a = first->next;
        ListNode* b = second->next;
        first->next = second;
        second->next = a;
        first = a;
        second = b;
    }
    return head;
}
