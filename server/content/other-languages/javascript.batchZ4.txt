//### valid-number-format
function validNumber(text) {
  const n = text.length;
  const isDigit = (c) => c >= "0" && c <= "9";
  let i = 0;
  if (i < n && (text[i] === "+" || text[i] === "-")) i++;
  let before = 0;
  while (i < n && isDigit(text[i])) { i++; before++; }
  let after = 0;
  if (i < n && text[i] === ".") {
    i++;
    while (i < n && isDigit(text[i])) { i++; after++; }
  }
  if (before + after === 0) return false;
  if (i < n && (text[i] === "e" || text[i] === "E")) {
    i++;
    if (i < n && (text[i] === "+" || text[i] === "-")) i++;
    let exponent = 0;
    while (i < n && isDigit(text[i])) { i++; exponent++; }
    if (exponent === 0) return false;
  }
  return i === n;
}
//### smallest-distinct-subsequence
function smallestDistinctSubsequence(text) {
  const last = new Map();
  for (let i = 0; i < text.length; i++) last.set(text[i], i);
  const stack = [], inStack = new Set();
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (inStack.has(c)) continue;
    while (stack.length > 0 && stack[stack.length - 1] > c && last.get(stack[stack.length - 1]) > i) {
      inStack.delete(stack.pop());
    }
    stack.push(c);
    inStack.add(c);
  }
  return stack.join("");
}
//### remove-k-digits
function removeKDigits(num, k) {
  const stack = [];
  for (const d of num) {
    while (k > 0 && stack.length > 0 && stack[stack.length - 1] > d) { stack.pop(); k--; }
    stack.push(d);
  }
  while (k > 0) { stack.pop(); k--; }
  const result = stack.join("").replace(/^0+/, "");
  return result === "" ? "0" : result;
}
//### pattern-132-exists
function has132Pattern(numbers) {
  let third = -Infinity;
  const stack = [];
  for (let i = numbers.length - 1; i >= 0; i--) {
    const x = numbers[i];
    if (x < third) return true;
    while (stack.length > 0 && stack[stack.length - 1] < x) third = stack.pop();
    stack.push(x);
  }
  return false;
}
//### sum-subarray-minimums
function sumSubarrayMins(numbers) {
  const MOD = 1000000007n, n = numbers.length;
  const left = new Array(n).fill(0), right = new Array(n).fill(0);
  let stack = [];
  for (let i = 0; i < n; i++) {
    while (stack.length > 0 && numbers[stack[stack.length - 1]] > numbers[i]) stack.pop();
    left[i] = i - (stack.length > 0 ? stack[stack.length - 1] : -1);
    stack.push(i);
  }
  stack = [];
  for (let i = n - 1; i >= 0; i--) {
    while (stack.length > 0 && numbers[stack[stack.length - 1]] >= numbers[i]) stack.pop();
    right[i] = (stack.length > 0 ? stack[stack.length - 1] : n) - i;
    stack.push(i);
  }
  let total = 0n;
  for (let i = 0; i < n; i++) total += BigInt(numbers[i]) * BigInt(left[i]) * BigInt(right[i]);
  return Number(total % MOD);
}
//### reverse-in-brackets
function reverseInBrackets(text) {
  const n = text.length, partner = new Array(n).fill(-1), opens = [];
  for (let i = 0; i < n; i++) {
    if (text[i] === "(") opens.push(i);
    else if (text[i] === ")") { const j = opens.pop(); partner[i] = j; partner[j] = i; }
  }
  const out = [];
  let i = 0, step = 1;
  while (i >= 0 && i < n) {
    if (text[i] === "(" || text[i] === ")") { i = partner[i]; step = -step; }
    else out.push(text[i]);
    i += step;
  }
  return out.join("");
}
//### car-fleet-count
function countCarFleets(target, position, speed) {
  const order = position.map((p, i) => i).sort((a, b) => position[b] - position[a]);
  let fleets = 0, leadDistance = 0, leadSpeed = 1;
  for (const i of order) {
    const distance = target - position[i];
    if (fleets === 0 || distance * leadSpeed > leadDistance * speed[i]) {
      fleets++;
      leadDistance = distance;
      leadSpeed = speed[i];
    }
  }
  return fleets;
}
//### remove-list-elements
function removeListValues(head, val) {
  const dummy = new ListNode(0, head);
  let prev = dummy;
  while (prev.next !== null) {
    if (prev.next.val === val) prev.next = prev.next.next;
    else prev = prev.next;
  }
  return dummy.next;
}
//### delete-middle-node
function deleteMiddle(head) {
  if (head === null || head.next === null) return null;
  let prev = null, slow = head, fast = head;
  while (fast !== null && fast.next !== null) {
    prev = slow;
    slow = slow.next;
    fast = fast.next.next;
  }
  prev.next = slow.next;
  return head;
}
//### reorder-list-alternating
function reorderAlternating(head) {
  if (head === null || head.next === null) return head;
  let slow = head, fast = head;
  while (fast.next !== null && fast.next.next !== null) { slow = slow.next; fast = fast.next.next; }
  let second = slow.next;
  slow.next = null;
  let prev = null;
  while (second !== null) { const nxt = second.next; second.next = prev; prev = second; second = nxt; }
  let first = head;
  second = prev;
  while (second !== null) {
    const a = first.next, b = second.next;
    first.next = second;
    second.next = a;
    first = a;
    second = b;
  }
  return head;
}
