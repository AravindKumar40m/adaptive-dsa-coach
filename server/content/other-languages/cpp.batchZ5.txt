//### remove-all-duplicate-values
ListNode* remove_all_duplicates(ListNode* head) {
    ListNode dummy(0);
    dummy.next = head;
    ListNode* prev = &dummy;
    ListNode* cur = head;
    while (cur != nullptr) {
        if (cur->next != nullptr && cur->next->val == cur->val) {
            int v = cur->val;
            while (cur != nullptr && cur->val == v) cur = cur->next;
            prev->next = cur;
        } else {
            prev = cur;
            cur = cur->next;
        }
    }
    return dummy.next;
}
//### double-digit-list
ListNode* double_digit_list(ListNode* head) {
    vector<int> digits;
    for (ListNode* node = head; node != nullptr; node = node->next) digits.push_back(node->val);
    int carry = 0;
    for (int i = (int) digits.size() - 1; i >= 0; i--) {
        int v = digits[i] * 2 + carry;
        digits[i] = v % 10;
        carry = v / 10;
    }
    if (carry > 0) digits.insert(digits.begin(), carry);
    ListNode dummy(0);
    ListNode* tail = &dummy;
    for (int d : digits) { tail->next = new ListNode(d); tail = tail->next; }
    return dummy.next;
}
//### min-depth
int tree_min_depth(TreeNode* root) {
    if (root == nullptr) return 0;
    queue<pair<TreeNode*, int>> q;
    q.push({root, 1});
    while (!q.empty()) {
        pair<TreeNode*, int> cur = q.front();
        q.pop();
        TreeNode* node = cur.first;
        if (node->left == nullptr && node->right == nullptr) return cur.second;
        if (node->left != nullptr) q.push({node->left, cur.second + 1});
        if (node->right != nullptr) q.push({node->right, cur.second + 1});
    }
    return 0;
}
//### sum-left-leaves
int left_leaves_walk(TreeNode* node) {
    if (node == nullptr) return 0;
    int total = 0;
    if (node->left != nullptr && node->left->left == nullptr && node->left->right == nullptr) total += node->left->val;
    else total += left_leaves_walk(node->left);
    return total + left_leaves_walk(node->right);
}

int sum_left_leaves(TreeNode* root) {
    return left_leaves_walk(root);
}
//### is-subtree
void subtree_text(TreeNode* node, string& out) {
    if (node == nullptr) { out += "#,"; return; }
    out += "^" + to_string(node->val) + ",";
    subtree_text(node->left, out);
    subtree_text(node->right, out);
}

bool is_subtree(TreeNode* root, TreeNode* sub) {
    string a = ",", b = ",";
    subtree_text(root, a);
    subtree_text(sub, b);
    return a.find(b) != string::npos;
}
//### longest-univalue-path
int univalue_down(TreeNode* node, int& best) {
    if (node == nullptr) return 0;
    int left = univalue_down(node->left, best), right = univalue_down(node->right, best);
    int left_arm = (node->left != nullptr && node->left->val == node->val) ? left + 1 : 0;
    int right_arm = (node->right != nullptr && node->right->val == node->val) ? right + 1 : 0;
    best = max(best, left_arm + right_arm);
    return max(left_arm, right_arm);
}

int longest_univalue_path(TreeNode* root) {
    int best = 0;
    univalue_down(root, best);
    return best;
}
//### flatten-to-right-chain
TreeNode* flatten_to_chain(TreeNode* root) {
    TreeNode* cur = root;
    while (cur != nullptr) {
        if (cur->left != nullptr) {
            TreeNode* last = cur->left;
            while (last->right != nullptr) last = last->right;
            last->right = cur->right;
            cur->right = cur->left;
            cur->left = nullptr;
        }
        cur = cur->right;
    }
    return root;
}
//### nodes-at-distance-k
int distance_collect(TreeNode* node, int depth) {
    if (node == nullptr) return 0;
    if (depth == 0) return 1;
    return distance_collect(node->left, depth - 1) + distance_collect(node->right, depth - 1);
}

int distance_go(TreeNode* node, int target, int k, int& found) {
    if (node == nullptr) return -1;
    if (node->val == target) { found += distance_collect(node, k); return 0; }
    int left = distance_go(node->left, target, k, found);
    if (left != -1) {
        int dist = left + 1;
        if (dist == k) found += 1;
        else if (dist < k) found += distance_collect(node->right, k - dist - 1);
        return dist;
    }
    int right = distance_go(node->right, target, k, found);
    if (right != -1) {
        int dist = right + 1;
        if (dist == k) found += 1;
        else if (dist < k) found += distance_collect(node->left, k - dist - 1);
        return dist;
    }
    return -1;
}

int count_nodes_at_distance(TreeNode* root, int target, int k) {
    int found = 0;
    distance_go(root, target, k, found);
    return found;
}
//### hanoi-move-count
int hanoi_move_count(int n) {
    return (1 << n) - 1;
}
//### strobogrammatic-count
int strobogrammatic_count(int n) {
    if (n == 1) return 3;
    int pairs = n / 2;
    long long result = n % 2 == 1 ? 3 : 1;
    for (int i = 0; i < pairs; i++) result *= (i == 0 ? 4 : 5);
    return (int) result;
}
