//### set-mismatch-pair
class Solution {
    public int[] findDuplicateMissing(int[] numbers) {
        int n = numbers.length;
        boolean[] seen = new boolean[n + 1];
        int repeated = 0;
        long total = 0;
        for (int x : numbers) {
            if (seen[x]) repeated = x; else seen[x] = true;
            total += x;
        }
        long missing = (long) n * (n + 1) / 2 - (total - repeated);
        return new int[]{repeated, (int) missing};
    }
}
//### running-balance-min-start
class Solution {
    public int minStartValue(int[] changes) {
        long prefix = 0, lowest = 0;
        for (int c : changes) {
            prefix += c;
            lowest = Math.min(lowest, prefix);
        }
        return (int) (1 - lowest);
    }
}
//### diagonal-traverse
class Solution {
    public int[] diagonalOrder(int[][] matrix) {
        int rows = matrix.length, cols = matrix[0].length;
        int[] out = new int[rows * cols];
        int k = 0;
        for (int d = 0; d < rows + cols - 1; d++) {
            int lo = Math.max(0, d - cols + 1), hi = Math.min(rows - 1, d);
            if (d % 2 == 0) {
                for (int r = hi; r >= lo; r--) out[k++] = matrix[r][d - r];
            } else {
                for (int r = lo; r <= hi; r++) out[k++] = matrix[r][d - r];
            }
        }
        return out;
    }
}
//### pivot-index-balance
class Solution {
    public int pivotIndex(int[] numbers) {
        long total = 0;
        for (int x : numbers) total += x;
        long left = 0;
        for (int i = 0; i < numbers.length; i++) {
            if (left == total - left - numbers[i]) return i;
            left += numbers[i];
        }
        return -1;
    }
}
//### lucky-numbers-in-matrix
class Solution {
    public int[] luckyValues(int[][] matrix) {
        int rows = matrix.length, cols = matrix[0].length;
        int[] rowMin = new int[rows], colMax = new int[cols];
        Arrays.fill(rowMin, Integer.MAX_VALUE);
        Arrays.fill(colMax, Integer.MIN_VALUE);
        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                rowMin[r] = Math.min(rowMin[r], matrix[r][c]);
                colMax[c] = Math.max(colMax[c], matrix[r][c]);
            }
        }
        List<Integer> out = new ArrayList<>();
        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                if (matrix[r][c] == rowMin[r] && matrix[r][c] == colMax[c]) out.add(matrix[r][c]);
            }
        }
        Collections.sort(out);
        int[] result = new int[out.size()];
        for (int i = 0; i < result.length; i++) result[i] = out.get(i);
        return result;
    }
}
//### shortest-unsorted-stretch
class Solution {
    public int unsortedStretchLength(int[] numbers) {
        int n = numbers.length, end = -1, start = 0;
        long biggest = Long.MIN_VALUE, smallest = Long.MAX_VALUE;
        for (int i = 0; i < n; i++) {
            if (numbers[i] < biggest) end = i; else biggest = numbers[i];
        }
        for (int i = n - 1; i >= 0; i--) {
            if (numbers[i] > smallest) start = i; else smallest = numbers[i];
        }
        return end - start + 1;
    }
}
//### increasing-triplet-exists
class Solution {
    public boolean hasIncreasingTriplet(int[] numbers) {
        long first = Long.MAX_VALUE, second = Long.MAX_VALUE;
        for (int x : numbers) {
            if (x <= first) first = x;
            else if (x <= second) second = x;
            else return true;
        }
        return false;
    }
}
//### relative-sort-order
class Solution {
    public int[] relativeSort(int[] values, int[] order) {
        Map<Integer, Integer> position = new HashMap<>();
        for (int i = 0; i < order.length; i++) position.put(order[i], i);
        Integer[] boxed = new Integer[values.length];
        for (int i = 0; i < values.length; i++) boxed[i] = values[i];
        Arrays.sort(boxed, (a, b) -> {
            int ra = position.getOrDefault(a, order.length), rb = position.getOrDefault(b, order.length);
            if (ra != rb) return Integer.compare(ra, rb);
            return Integer.compare(a, b);
        });
        int[] result = new int[values.length];
        for (int i = 0; i < result.length; i++) result[i] = boxed[i];
        return result;
    }
}
//### digit-sum-sort
class Solution {
    private int digitSum(int x) {
        int total = 0;
        while (x > 0) { total += x % 10; x /= 10; }
        return total;
    }

    public int[] sortByDigitSum(int[] numbers) {
        Integer[] boxed = new Integer[numbers.length];
        for (int i = 0; i < numbers.length; i++) boxed[i] = numbers[i];
        Arrays.sort(boxed, (a, b) -> {
            int da = digitSum(a), db = digitSum(b);
            if (da != db) return Integer.compare(da, db);
            return Integer.compare(a, b);
        });
        int[] result = new int[numbers.length];
        for (int i = 0; i < result.length; i++) result[i] = boxed[i];
        return result;
    }
}
//### minimum-swaps-to-sort
class Solution {
    public int minSwapsToSort(int[] numbers) {
        int n = numbers.length;
        Integer[] order = new Integer[n];
        for (int i = 0; i < n; i++) order[i] = i;
        Arrays.sort(order, (a, b) -> Integer.compare(numbers[a], numbers[b]));
        int[] target = new int[n];
        for (int rank = 0; rank < n; rank++) target[order[rank]] = rank;
        boolean[] visited = new boolean[n];
        int cycles = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                cycles++;
                int j = i;
                while (!visited[j]) { visited[j] = true; j = target[j]; }
            }
        }
        return n - cycles;
    }
}
