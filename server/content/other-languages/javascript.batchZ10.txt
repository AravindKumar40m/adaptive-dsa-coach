//### stock-profit-with-fee
function stockProfitWithFee(prices, fee) {
  let cash = 0, hold = -1000000000;
  for (const price of prices) {
    cash = Math.max(cash, hold + price - fee);
    hold = Math.max(hold, cash - price);
  }
  return cash;
}
//### maximal-rectangle-of-ones
function maxRectangleOfOnes(grid) {
  const rows = grid.length, cols = grid[0].length;
  const heights = new Array(cols).fill(0);
  let best = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) heights[c] = grid[r][c] === 1 ? heights[c] + 1 : 0;
    const stack = [];
    for (let i = 0; i <= cols; i++) {
      const h = i < cols ? heights[i] : 0;
      while (stack.length > 0 && heights[stack[stack.length - 1]] >= h) {
        const top = stack.pop();
        const width = stack.length === 0 ? i : i - stack[stack.length - 1] - 1;
        best = Math.max(best, heights[top] * width);
      }
      stack.push(i);
    }
  }
  return best;
}
//### dungeon-min-health
function minInitialHealth(dungeon) {
  const rows = dungeon.length, cols = dungeon[0].length;
  const INF = 1000000000;
  const need = new Array(cols + 1).fill(INF);
  need[cols - 1] = 1;
  for (let r = rows - 1; r >= 0; r--) {
    for (let c = cols - 1; c >= 0; c--) {
      let next = r < rows - 1 ? need[c] : INF;
      if (c + 1 < cols) next = Math.min(next, need[c + 1]);
      if (r === rows - 1 && c === cols - 1) next = 1;
      need[c] = Math.max(1, next - dungeon[r][c]);
    }
  }
  return need[0];
}
//### two-resource-knapsack
function maxItemsTwoBudgets(items, zeros, ones) {
  const best = [];
  for (let a = 0; a <= zeros; a++) best.push(new Array(ones + 1).fill(0));
  for (const [zc, oc] of items) {
    for (let a = zeros; a >= zc; a--) {
      for (let b = ones; b >= oc; b--) best[a][b] = Math.max(best[a][b], best[a - zc][b - oc] + 1);
    }
  }
  return best[zeros][ones];
}
//### interleaving-strings-check
function isInterleaving(a, b, c) {
  if (a.length + b.length !== c.length) return false;
  const m = a.length, n = b.length;
  const row = new Array(n + 1).fill(false);
  row[0] = true;
  for (let j = 1; j <= n; j++) row[j] = row[j - 1] && b[j - 1] === c[j - 1];
  for (let i = 1; i <= m; i++) {
    row[0] = row[0] && a[i - 1] === c[i - 1];
    for (let j = 1; j <= n; j++) row[j] = (row[j] && a[i - 1] === c[i + j - 1]) || (row[j - 1] && b[j - 1] === c[i + j - 1]);
  }
  return row[n];
}
//### distinct-subsequence-count
function countDistinctSubsequences(s, t) {
  const m = t.length;
  const ways = new Array(m + 1).fill(0);
  ways[0] = 1;
  for (const ch of s) {
    for (let j = m; j >= 1; j--) if (ch === t[j - 1]) ways[j] += ways[j - 1];
  }
  return ways[m];
}
//### min-falling-path-sum
function minFallingPathSum(matrix) {
  const rows = matrix.length, cols = matrix[0].length;
  let best = matrix[0].slice();
  for (let r = 1; r < rows; r++) {
    const next = [];
    for (let c = 0; c < cols; c++) {
      let low = best[c];
      if (c > 0) low = Math.min(low, best[c - 1]);
      if (c + 1 < cols) low = Math.min(low, best[c + 1]);
      next.push(matrix[r][c] + low);
    }
    best = next;
  }
  return Math.min(...best);
}
//### domino-tromino-tilings
function countTilings(n) {
  const MOD = 1000000007;
  if (n <= 2) return [1, 1, 2][n];
  let a = 1, b = 1, c = 2;
  for (let i = 3; i <= n; i++) { const next = (2 * c + a) % MOD; a = b; b = c; c = next; }
  return c;
}
//### regex-star-match
function regexStarMatch(text, pattern) {
  const n = text.length, m = pattern.length;
  const table = [];
  for (let i = 0; i <= n; i++) table.push(new Array(m + 1).fill(false));
  table[0][0] = true;
  for (let j = 1; j <= m; j++) if (pattern[j - 1] === "*") table[0][j] = table[0][j - 2];
  for (let i = 1; i <= n; i++) {
    for (let j = 1; j <= m; j++) {
      if (pattern[j - 1] === "*") {
        table[i][j] = table[i][j - 2] || ((pattern[j - 2] === "." || pattern[j - 2] === text[i - 1]) && table[i - 1][j]);
      } else {
        table[i][j] = (pattern[j - 1] === "." || pattern[j - 1] === text[i - 1]) && table[i - 1][j - 1];
      }
    }
  }
  return table[n][m];
}
//### ascii-delete-sum
function minAsciiDeleteSum(s1, s2) {
  const n = s1.length, m = s2.length;
  const keep = new Array(m + 1).fill(0);
  for (let i = 1; i <= n; i++) {
    let previousDiagonal = 0;
    for (let j = 1; j <= m; j++) {
      const saved = keep[j];
      if (s1[i - 1] === s2[j - 1]) keep[j] = previousDiagonal + s1.charCodeAt(i - 1);
      else keep[j] = Math.max(keep[j], keep[j - 1]);
      previousDiagonal = saved;
    }
  }
  let total = 0;
  for (const ch of s1) total += ch.charCodeAt(0);
  for (const ch of s2) total += ch.charCodeAt(0);
  return total - 2 * keep[m];
}
