//### bouquet-days
function bouquetDays(bloom, m, k) {
  if (m * k > bloom.length) return -1;
  function enough(day) {
    let run = 0, bouquets = 0;
    for (const b of bloom) {
      if (b <= day) {
        run++;
        if (run === k) { bouquets++; run = 0; }
      } else run = 0;
    }
    return bouquets >= m;
  }
  let lo = bloom[0], hi = bloom[0];
  for (const b of bloom) { if (b < lo) lo = b; if (b > hi) hi = b; }
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (enough(mid)) hi = mid; else lo = mid + 1;
  }
  return lo;
}
//### min-rotated-with-duplicates
function minRotatedDuplicates(numbers) {
  let lo = 0, hi = numbers.length - 1;
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (numbers[mid] > numbers[hi]) lo = mid + 1;
    else if (numbers[mid] < numbers[hi]) hi = mid;
    else hi--;
  }
  return numbers[lo];
}
//### kth-smallest-pair-distance
function kthPairDistance(numbers, k) {
  const a = numbers.slice().sort((p, q) => p - q);
  function count(d) {
    let j = 0, total = 0;
    for (let i = 0; i < a.length; i++) {
      while (a[i] - a[j] > d) j++;
      total += i - j;
    }
    return total;
  }
  let lo = 0, hi = a[a.length - 1] - a[0];
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (count(mid) >= k) hi = mid; else lo = mid + 1;
  }
  return lo;
}
//### spread-out-stalls
function spreadStalls(positions, animals) {
  const a = positions.slice().sort((p, q) => p - q);
  function fits(d) {
    let placed = 1, last = a[0];
    for (let i = 1; i < a.length; i++) {
      if (a[i] - last >= d) { placed++; last = a[i]; }
    }
    return placed >= animals;
  }
  let lo = 1, hi = a[a.length - 1] - a[0];
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo + 1) / 2);
    if (fits(mid)) lo = mid; else hi = mid - 1;
  }
  return lo;
}
//### search-in-bitonic-array
function searchBitonic(numbers, target) {
  const n = numbers.length;
  if (n === 0) return -1;
  let lo = 0, hi = n - 1;
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (numbers[mid] < numbers[mid + 1]) lo = mid + 1; else hi = mid;
  }
  const peak = lo;
  lo = 0; hi = peak;
  while (lo <= hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (numbers[mid] === target) return mid;
    if (numbers[mid] < target) lo = mid + 1; else hi = mid - 1;
  }
  lo = peak + 1; hi = n - 1;
  while (lo <= hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (numbers[mid] === target) return mid;
    if (numbers[mid] > target) lo = mid + 1; else hi = mid - 1;
  }
  return -1;
}
//### k-smallest-pair-sums
function heapPush(a, x, less) {
  a.push(x);
  let i = a.length - 1;
  while (i > 0) {
    const p = (i - 1) >> 1;
    if (less(a[i], a[p])) { const t = a[i]; a[i] = a[p]; a[p] = t; i = p; } else break;
  }
}
function heapPop(a, less) {
  const top = a[0];
  const last = a.pop();
  if (a.length > 0) {
    a[0] = last;
    let i = 0;
    for (;;) {
      const l = 2 * i + 1, r = l + 1;
      let m = i;
      if (l < a.length && less(a[l], a[m])) m = l;
      if (r < a.length && less(a[r], a[m])) m = r;
      if (m === i) break;
      const t = a[i]; a[i] = a[m]; a[m] = t;
      i = m;
    }
  }
  return top;
}
function kSmallestPairSums(first, second, k) {
  const out = [];
  if (first.length === 0 || second.length === 0) return out;
  const less = (p, q) => p[0] < q[0];
  const heap = [];
  for (let i = 0; i < Math.min(first.length, k); i++) heapPush(heap, [first[i] + second[0], i, 0], less);
  while (heap.length > 0 && out.length < k) {
    const [total, i, j] = heapPop(heap, less);
    out.push(total);
    if (j + 1 < second.length) heapPush(heap, [first[i] + second[j + 1], i, j + 1], less);
  }
  return out;
}
//### k-closest-elements
function kClosestElements(numbers, k, x) {
  let lo = 0, hi = numbers.length - k;
  while (lo < hi) {
    const mid = lo + Math.floor((hi - lo) / 2);
    if (x - numbers[mid] > numbers[mid + k] - x) lo = mid + 1; else hi = mid;
  }
  return numbers.slice(lo, lo + k);
}
//### task-cooldown-time
function taskCooldownTime(tasks, cooldown) {
  if (tasks.length === 0) return 0;
  const counts = new Array(26).fill(0);
  for (const ch of tasks) counts[ch.charCodeAt(0) - 65]++;
  let top = 0;
  for (const c of counts) top = Math.max(top, c);
  let numTop = 0;
  for (const c of counts) if (c === top) numTop++;
  return Math.max(tasks.length, (top - 1) * (cooldown + 1) + numTop);
}

function maxCapital(k, w, profits, capital) {
  const projects = profits.map((p, i) => [capital[i], p]).sort((a, b) => a[0] - b[0]);
  const heap = [];
  const bigger = (p, q) => p > q;
  let i = 0;
  for (let step = 0; step < k; step++) {
    while (i < projects.length && projects[i][0] <= w) { heapPush(heap, projects[i][1], bigger); i++; }
    if (heap.length === 0) break;
    w += heapPop(heap, bigger);
  }
  return w;
}
//### smallest-covering-range
function heapPush(a, x, less) {

      if (r < a.length && less(a[r], a[m])) m = r;

function smallestCoveringRange(lists) {
  const less = (p, q) => p[0] < q[0];
  const heap = [];
  let high = lists[0][0];
  for (let r = 0; r < lists.length; r++) {
    heapPush(heap, [lists[r][0], r, 0], less);
    high = Math.max(high, lists[r][0]);
  }
  let best = [heap[0][0], high];
  for (;;) {
    const [low, r, c] = heapPop(heap, less);
    if (high - low < best[1] - best[0]) best = [low, high];
    if (c + 1 === lists[r].length) break;
    const next = lists[r][c + 1];
    high = Math.max(high, next);
    heapPush(heap, [next, r, c + 1], less);
  }
  return best;
}
